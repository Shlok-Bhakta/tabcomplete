from __future__ import annotations
import hashlib
import json
import os
import sys
import time
from pathlib import Path

REPO = Path('/home/crabcake/Projects/tabcomplete-product-r2')
sys.path.insert(0, str(REPO / 'scripts'))
from evaluate_one_line_local import sha256_file, resident_bytes
from tinycomplete.one_line.contract import EditState, EditAction, decode_action
from tinycomplete.one_line.train import encode_training_row

plan_path = REPO / 'reports/prototype/product_r2/disposable_fixture_export_reload_plan_v1.json'
plan = json.loads(plan_path.read_text())
assert sha256_file(plan_path) == sys.argv[1]
for name, digest in plan['source_files'].items():
    assert sha256_file(REPO / name) == digest
assert sha256_file(Path(__file__)) == plan['reload_checker_sha256']
model_dir = Path(plan['model_dir'])
assert sha256_file(model_dir / 'model.safetensors') == plan['weight_sha256']
assert sha256_file(model_dir / 'tokenizer.json') == plan['tokenizer_sha256']
fixture = Path(plan['fixture_path'])
assert sha256_file(fixture) == plan['fixture_sha256']
rows = [json.loads(line) for line in fixture.read_text().splitlines()]
assert len(rows) == 64
assert all(r['quality_evidence'] is False for r in rows)
os.environ.update(HF_HUB_OFFLINE='1', TRANSFORMERS_OFFLINE='1')
import torch
import transformers
from transformers import AutoTokenizer, AutoModelForCausalLM
assert torch.__version__ == plan['torch']
assert transformers.__version__ == plan['transformers']
assert not torch.cuda.is_available()
torch.set_num_threads(plan['threads'])
torch.set_num_interop_threads(plan['interop_threads'])
tokenizer = AutoTokenizer.from_pretrained(model_dir, local_files_only=True, trust_remote_code=False)
source_tokenizer = AutoTokenizer.from_pretrained(plan['source_tokenizer_dir'], local_files_only=True, trust_remote_code=False)
assert tokenizer.eos_token_id == source_tokenizer.eos_token_id
assert tokenizer.get_vocab() == source_tokenizer.get_vocab()
encoded = {}
for row in rows:
    state, action = EditState.from_mapping(row['state']), EditAction(**row['action'])
    example = encode_training_row(tokenizer, state, action)
    source_example = encode_training_row(source_tokenizer, state, action)
    assert example == source_example
    encoded[row['id']] = example
started = time.perf_counter()
model = AutoModelForCausalLM.from_pretrained(model_dir, local_files_only=True, trust_remote_code=False, dtype=torch.float32, low_cpu_mem_usage=True)
model.eval()
load_seconds = time.perf_counter() - started
parameters = sum(p.numel() for p in model.parameters())
assert parameters == 494032768
observations = []
by_id = {row['id']: row for row in rows}
for identifier in plan['probe_ids']:
    example = encoded[identifier]
    ids = torch.tensor([example.input_ids[:example.prompt_tokens]])
    started = time.perf_counter()
    with torch.inference_mode():
        generated = model.generate(ids, attention_mask=torch.ones_like(ids), max_new_tokens=64, do_sample=False, use_cache=True, eos_token_id=tokenizer.eos_token_id, pad_token_id=tokenizer.eos_token_id)[0, ids.shape[1]:].tolist()
    elapsed = time.perf_counter() - started
    terminated = bool(generated and generated[-1] == tokenizer.eos_token_id)
    wire = tokenizer.decode(generated[:-1] if terminated else generated, skip_special_tokens=False, clean_up_tokenization_spaces=False)
    decoded = decode_action(wire, terminated=terminated, generated_tokens=len(generated))
    observations.append({'id': identifier, 'input_tokens': ids.shape[1], 'generated_tokens': len(generated), 'terminated_by_eos': terminated, 'valid_action': decoded.action is not None, 'exact_action': decoded.action == EditAction(**by_id[identifier]['action']), 'wire': wire, 'elapsed_seconds': elapsed, 'resident_bytes_after': resident_bytes()})
result = {'schema': 'disposable-export-reload-observation-v1', 'plan_sha256': sys.argv[1], 'pid': os.getpid(), 'runtime': {'torch': torch.__version__, 'transformers': transformers.__version__, 'backend': plan['backend'], 'threads': torch.get_num_threads()}, 'load_seconds': load_seconds, 'parameters': parameters, 'tied_embeddings': model.get_input_embeddings().weight.data_ptr() == model.get_output_embeddings().weight.data_ptr(), 'tokenizer_parity_cases': len(rows), 'observations': observations, 'quality_evidence': False, 'gpu_resume_execution_verified': False}
Path(sys.argv[2]).write_text(json.dumps(result, sort_keys=True, indent=2) + '\n')
print(json.dumps({'observations': len(observations), 'exact_eos': sum(r['exact_action'] and r['terminated_by_eos'] for r in observations), 'pid': os.getpid(), 'load_seconds': load_seconds}))
assert all(r['exact_action'] and r['terminated_by_eos'] for r in observations)
